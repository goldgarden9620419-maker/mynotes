# 하이웍스 받은파일 자동정리 스크립트 (Google Apps Script)

> 2026-10-23 작성. claude.ai 주간 루틴의 승인 창 문제를 해결하기 위해 Google Drive 내부 자동화(Apps Script)로 이전.
> 대상 폴더: 하이웍스 받은파일 (folder id: `1DaJLs5UQAp5tj2ke8rHe8A-_nSqVs38x`)
> 실행 주기: 매주 금요일 오후 6~7시 (시간 기반 트리거)

## 스크립트 코드

```javascript
// ===== 하이웍스 받은파일 주간 자동정리 =====
const ROOT_FOLDER_ID = '1DaJLs5UQAp5tj2ke8rHe8A-_nSqVs38x';
const SEND_EMAIL = true; // 정리 후 이메일 보고 (끄려면 false)

// 파일명 키워드 → 폴더 분류 규칙 (위에서부터 먼저 일치하는 규칙 적용)
const RULES = [
  { keywords: ['거래명세서', '견적'],        folder: '거래명세서_견적서' },
  { keywords: ['업무일지', '업무계획'],      folder: '업무일지_업무계획' },
  { keywords: ['택배현황'],                  folder: '택배현황' },
  { keywords: ['세금계산서', '지출목록'],    folder: '회계_세무_정산' },
  { keywords: ['주문장', '판촉'],            folder: '판촉주문장_주문장취합' },
  { keywords: ['쿠팡로켓배송', '입고관리'],  folder: '쿠팡로켓배송_입고관리' },
  { keywords: ['지원사업', 'PoC'],           folder: '지원사업' },
  { keywords: ['약령몰'],                    folder: '약령몰' },
];
const FALLBACK_FOLDER = '기타'; // 어느 규칙에도 안 맞는 파일

function weeklyCleanup() {
  const root = DriveApp.getFolderById(ROOT_FOLDER_ID);
  const files = root.getFiles();
  const moved = [];

  while (files.hasNext()) {
    const file = files.next();
    const name = file.getName();
    let target = FALLBACK_FOLDER;
    for (const rule of RULES) {
      if (rule.keywords.some(k => name.toLowerCase().includes(k.toLowerCase()))) {
        target = rule.folder;
        break;
      }
    }
    file.moveTo(getOrCreateSubfolder(root, target));
    moved.push(name + '  →  ' + target);
  }

  if (SEND_EMAIL && moved.length > 0) {
    MailApp.sendEmail(
      Session.getActiveUser().getEmail(),
      '[하이웍스 받은파일] 주간 정리 완료 — ' + moved.length + '건',
      '이번 주 정리 내역:\n\n' + moved.join('\n')
    );
  }
}

function getOrCreateSubfolder(root, name) {
  const it = root.getFoldersByName(name);
  return it.hasNext() ? it.next() : root.createFolder(name);
}
```

## 설치 방법

1. https://script.google.com 접속 (goldgarden9620419@gmail.com 계정)
2. **새 프로젝트** → 기본 코드를 지우고 위 코드 전체 붙여넣기 → 프로젝트 이름 "하이웍스 폴더 정리"로 저장
3. 상단 함수 선택을 `weeklyCleanup`으로 두고 **실행(▶)** 1회 → 권한 승인 (최초 1회, "확인되지 않은 앱" 경고가 떠도 본인 스크립트이므로 "고급 → 이동"으로 승인)
4. 왼쪽 **시계 아이콘(트리거) → 트리거 추가**: 함수 `weeklyCleanup`, 이벤트 소스 "시간 기반", "주 단위 타이머", "매주 금요일", "오후 6시~7시" → 저장

## 참고

- 정리 결과는 매주 이메일로 발송됨 (`SEND_EMAIL = false`로 끌 수 있음)
- 새 문서 종류 추가 시 `RULES` 배열에 한 줄 추가
